Tutorial 10: PK/PD Modelling

Simultaneous fitting of pharmacokinetics and pharmacodynamics

Introduction

Pharmacokinetic (PK) models describe how drug concentrations change over time. Pharmacodynamic (PD) models describe the relationship between concentration and effect. Simultaneous PK/PD modelling fits both systems together, allowing the neural network to predict individual parameters for both.

This tutorial demonstrates a PK/PD link model using:

  • PK: two-compartment with IV infusion
  • PD link: effect-compartment with equilibration rate ke0
  • PD effect: Emax sigmoid model — E = Emax * Ce^γ / (EC50^γ + Ce^γ)

Both PK concentrations and PD effects are observed simultaneously. The DCM predicts all PK and PD parameters from patient covariates.

Data: Simulated to allow ground-truth validation. The true parameters vary realistically across subjects.

1 — Simulate a PK/PD population

We simulate 30 subjects with two observed endpoints:

  • y_pk: plasma concentrations (compartment 1)
  • y_pd: pharmacodynamic effect (derived from effect-site concentration)
# True PK/PD ODE: two-compartment PK + effect compartment
function pkpd_true!(dA, A, p, t)
    CL, V₁, Q, V₂, ke0, I = p
    k₁₀ = CL / V₁;  k₁₂ = Q / V₁;  k₂₁ = Q / V₂
    dA[1] = (I / V₁) + A[2] * k₂₁ - A[1] * (k₁₀ + k₁₂)   # plasma
    dA[2] = A[1] * k₁₂ - A[2] * k₂₁                         # peripheral
    dA[3] = ke0 * (A[1]/V₁ - A[3])                          # effect site Ce
end

# True PD: sigmoid Emax (fixed across population for simplicity)
Emax  = 100f0
EC50  = 3f0
gamma = 2f0
emax_pd(Ce) = Emax * Ce^gamma / (EC50^gamma + Ce^gamma)

function simulate_pkpd(id, weight, dose_mg, t_pk, t_pd, ::Type{T} = Float32) where T
    # True parameters: allometric scaling from weight
    CL   = T(2.5  * (weight/70)^0.75) * exp(T(0.3) * randn(rng, T))
    V₁   = T(5.0  * (weight/70))      * exp(T(0.2) * randn(rng, T))
    Q    = T(1.2)                      * exp(T(0.2) * randn(rng, T))
    V₂   = T(8.0  * (weight/70))
    ke0  = T(0.5)                      * exp(T(0.15) * randn(rng, T))

    p_true = T[CL, V₁, Q, V₂, ke0, 0f0]

    # 30-minute IV infusion: rate = dose/0.5h
    dur = T(0.5)
    A_dose = T[0f0  dose_mg  (dose_mg/dur)  dur]
    cb = generate_dosing_callback(A_dose, T)

    t_all = sort(unique(vcat(t_pk, t_pd)))
    prob  = ODEProblem(pkpd_true!, zeros(T, 3), (T(-0.1), maximum(t_all) + T(1)), p_true)
    sol   = solve(prob, Tsit5();
        callback = cb, saveat = t_all, tstops = cb.condition.times)

    # Extract PK (plasma = A[1]/V₁) and PD (Emax from Ce = A[3])
    sol_arr = Array(sol)
    t_idx_pk = indexin(t_pk, t_all)
    t_idx_pd = indexin(t_pd, t_all)

    y_pk = (sol_arr[1, t_idx_pk] ./ V₁) .+
           T(0.2) .* (sol_arr[1, t_idx_pk] ./ V₁) .* randn(rng, T, length(t_pk))
    y_pd = emax_pd.(sol_arr[3, t_idx_pd]) .+
           T(3.0) .* randn(rng, T, length(t_pd))

    y_pk = max.(y_pk, T(0))
    y_pd = clamp.(y_pd, T(0), T(100))

    return id, T[weight], t_pk, t_pd, y_pk, y_pd, cb, T
end

# PK samples: dense early, sparse late
t_pk = Float32[0.25, 0.5, 1.0, 2.0, 4.0, 8.0, 12.0, 24.0]
# PD samples: at different time points than PK
t_pd = Float32[0.5, 1.5, 3.0, 6.0, 10.0, 16.0]

weights = Float32.(60:5:110)   # 11 weight values, 3 subjects each
sim_data = [simulate_pkpd(i, weights[mod(i-1,11)+1], 200f0, t_pk, t_pd)
            for i in 1:30]

println("PK obs per subject: $(length(t_pk))")
println("PD obs per subject: $(length(t_pd))")

2 — Build an MOIndividual population

MOIndividual handles multiple response variables per subject. It takes:

  • ts: a Vector{Vector} of observation times — one per response
  • ys: a Vector{Vector} of observations — one per response

Internally it merges the time vectors into a single union grid and builds a dvid mask that records which time points belong to which response.

function make_mo_individual(id, x, t_pk, t_pd, y_pk, y_pd, cb, ::Type{T}) where T
    MOIndividual(
        id,
        Vector{T}(x),
        [Vector{T}(t_pk), Vector{T}(t_pd)],   # ts: [pk times, pd times]
        [Vector{T}(y_pk), Vector{T}(y_pd)],   # ys: [pk obs,   pd obs  ]
        cb,
        T,
    )
end

mo_pop = Population([make_mo_individual(d...) for d in sim_data])

3 — PK/PD ODE and DCM specification

function pkpd_ode!(dA, A, p, t)
    CL, V₁, Q, V₂, ke0, I = unpack(p, t)
    k₁₀ = CL / V₁;  k₁₂ = Q / V₁;  k₂₁ = Q / V₂
    dA[1] = (I / V₁) + A[2] * k₂₁ - A[1] * (k₁₀ + k₁₂)
    dA[2] = A[1] * k₁₂ - A[2] * k₂₁
    dA[3] = ke0 * (A[1]/V₁ - A[3])
end

# Neural network: weight → [CL, V₁, Q, V₂, ke0]
ann_pkpd = Lux.Chain(
    Normalize([120f0]),
    Lux.Dense(1, 16, tanh),
    Lux.Dense(16, 5, Lux.softplus),
)

# ErrorModelSet: separate error models for PK (proportional) and PD (additive)
errors = ErrorModelSet(ProportionalError(0.2f0), AdditiveError(5f0))

# target=[1,3]: observe plasma concentration (A[1]/V[1]) and effect site (A[3])
# (DCM returns raw amounts; we handle the PD transformation below)
model_pkpd = DCM(pkpd_ode!, 3, ann_pkpd, errors; target=[1, 3])

println("Model: $(model_pkpd)")
Model: DeepCompartmentModel{ode = pkpd_ode!, error = ErrorModelSet(ProportionalError, AdditiveError)}
Note

target=[1,3] tells the DCM to return compartments 1 and 3. For the PK response we want concentration A[1]/V₁, but predict returns raw amounts. The Emax PD transformation is applied post-prediction (see GOF section below). In a full implementation, consider using a CustomError that incorporates the nonlinear PD transformation inside the likelihood.

For MOIndividual, predict returns a Vector{Vector} — one concentration vector per response.

# ŷ is a 2-element vector: [PK concentrations, PD effect-site amounts]

# Collect GOF data across population
DV_pk, PRED_pk, DV_pd, PRED_pd = Float32[], Float32[], Float32[], Float32[]

for (i, indv) in enumerate(mo_pop)
    ps_i, st_i = take_batch(ps_pkpd, st_pkpd, [i])
    ŷ = predict(model_pkpd, indv, ps_i, st_i)
    # ŷ[1] = plasma amounts; concentration = amount / V₁
    V₁_pred = predict_typ_parameters(model_pkpd, indv, ps_pkpd, st_pkpd)[1][2]  # index 2 = V₁
    pk_conc  = ŷ[1] ./ V₁_pred
    pd_pred  = emax_pd.(ŷ[2])   # apply Emax transformation to effect-site amounts

    append!(DV_pk,   indv.ys[1])
    append!(PRED_pk, pk_conc)
    append!(DV_pd,   indv.ys[2])
    append!(PRED_pd, pd_pred)
end
